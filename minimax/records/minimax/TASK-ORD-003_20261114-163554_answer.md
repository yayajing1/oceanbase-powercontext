def filter_orders(
    orders: Iterable[Order],
    *,
    status: Optional[str] = None,
    category: Optional[str] = None,
) -> List[Order]:
    """Return the orders that match ``status`` and ``category``.

    ``status`` and ``category`` are optional; when one of them is ``None``
    the corresponding condition is not applied.  When both are given the
    conditions are combined with ``AND``.  The relative order of the
    returned orders always follows the input order.
    """
    matched: List[Order] = []
    for order in orders:
        if status is not None and order.status != status:
            continue
        if category is not None and order.category != category:
            continue
        matched.append(order)
    return matched
```

def to_decimal(value: object) -> Decimal:
    """Convert ``value`` to :class:`decimal.Decimal`.

    ``str`` inputs are parsed as decimal strings so that ``"10.50"`` and
    ``Decimal("10.50")`` compare equal.  Anything that cannot be parsed
    raises :class:`TypeError`.
    """
    if isinstance(value, Decimal):
        return value
    try:
        return Decimal(str(value))
    except (InvalidOperation, ValueError) as exc:
        raise TypeError(f"amount is not a valid decimal: {value!r}") from exc
```
